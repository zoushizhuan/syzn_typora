# 碰撞检测参数

```python
grasps_collision_param {
  num_grasps_per_batch: 300 返回有效抓取点的数量，越大耗时越久
  bin_collision_check_by_depth_map: false 如果为真，将使用深度图检查与箱子的碰撞。如果为假，将使用平面检查与箱子的碰撞。  如果箱壁不是矩形的，或者箱子的形状不是静态的（例如，箱壁偶尔变形）， 您可能还想将此设置为真，如果箱体模型（白线）与箱体之间没有紧密对接， 导致检测到错误的碰撞
  remove_obj_pc_threshold: 3 移出抓取工件的部分点云
  gripper_min_buffer: 5 夹爪与非抓取工件的最小距离
  instance_sampling_distance: 20 在进攻和撤退的运动过程中，检查夹爪与其他工件之间的碰撞
  pc_to_bin_sides_distance_threshold: 3 夹爪与箱壁距离阈值
  num_collision_pixel_threshold: 30 发生碰撞的像素阈值，如果发生碰撞的像素数量大于阈值，这被过滤
  retreat_collision_score_scale: 0
  fail_gripper_object_collisions: true
  detailed_logging: false 
  pc_to_bin_bottom_distance_threshold: 2
  downsampling_voxel_size_for_bin_collision_check: 10
  pointcloud_downsampling_ratios_for_scan_collision_check: 1
  early_stop_grasp_num: 30
  multithread: true
  debug_vis: false
  debug_vis_bin_collision: false
  debug_vis_scan_collision: false
  num_occlusion_pixel_threshold: 500
  gripper_collision_score_max_distance: 3
  gripper_depth_dilation_radius_pixels: 1
  num_missing_pixel_threshold: 1000
  num_implausible_pixel_threshold: 1000
  scan_dilation_radius_for_is_target_obj: 0
  scan_dilation_radius_for_gripper: 0
  scan_dilation_radius_for_target_obj_collision: 0
  scan_dilation_radius_for_missing_pixels: 0
  scan_dilation_radius_for_implausible_pixels: 0
  vectorized_pc2pc: true
  fail_gripper_object_collisions_fallback: true
  input_bin: true
  input_depth: false
  object_occlusion_score_min_value: 0
  filter_robot_bin_collisions: true
}
```
